import java.util.Scanner;

public class SalesTracker {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

        // Variables
        int maxTransactions;
        int transactionCounter = 0;
        int validTransactions = 0;
        int skippedTransactions = 0;
        int continueChoice;

        double totalSales = 0;
        double salesAmount;

        // Ask for maximum transactions
        System.out.print("Enter maximum number of sales transactions: ");
        maxTransactions = input.nextInt();

        // While loop
        while (transactionCounter < maxTransactions) {

            transactionCounter++;

            System.out.println("\nTransaction Number: " + transactionCounter);
            System.out.print("Enter sales amount: ");
            salesAmount = input.nextDouble();

            // Skip negative sales amounts
            if (salesAmount < 0) {
                System.out.println("Invalid sales amount. Transaction skipped.");
                skippedTransactions++;
                continue;
            }

            // Process valid transactions
            totalSales += salesAmount;
            validTransactions++;

            // Ask whether to continue
            if (transactionCounter < maxTransactions) {

                System.out.print("Continue processing? (1 = Yes, 0 = No): ");
                continueChoice = input.nextInt();

                if (continueChoice == 0) {
                    break;
                }
            }
        }

        // Check sales target
        System.out.println("\n========== SALES REPORT ==========");

        if (totalSales >= 10000) {
            System.out.println("Sales Target: Target Reached");
        } else {
            System.out.println("Sales Target: Target Not Reached");
        }

        // Display results
        System.out.printf("Total Sales Amount: ₱%.2f%n", totalSales);
        System.out.println("Valid Transactions: " + validTransactions);
        System.out.println("Skipped Transactions: " + skippedTransactions);

        // Close scanner
        input.close();
    }
}
